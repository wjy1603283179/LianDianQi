cmake_minimum_required(VERSION 3.20)
project(LianDianQi VERSION 1.0.2 LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "LianDianQi currently supports Windows only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Network)
add_library(clicker_core STATIC src/model.cpp src/model.h src/input.cpp src/input.h src/engine.cpp src/engine.h src/controls.cpp src/controls.h)
target_link_libraries(clicker_core PUBLIC Qt6::Widgets Qt6::Network user32)
target_include_directories(clicker_core PUBLIC src)
target_compile_definitions(clicker_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
add_executable(LianDianQi WIN32 src/main.cpp src/window.cpp src/window.h assets/app.rc assets/resources.qrc)
target_link_libraries(LianDianQi PRIVATE clicker_core)
if(MSVC)
    target_compile_options(clicker_core PUBLIC /utf-8 /W4)
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(clicker_tests tests/test_clicker.cpp src/window.cpp src/window.h assets/resources.qrc)
    target_link_libraries(clicker_tests PRIVATE clicker_core Qt6::Test)
    add_test(NAME clicker_tests COMMAND clicker_tests -o test-results.txt,txt)
    set_tests_properties(clicker_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS LianDianQi RUNTIME DESTINATION .)
